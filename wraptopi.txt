float wrapToPi(float angle) {
  while (angle > PI) {
    angle -= 2.0 * PI;
  }

  while (angle < -PI) {
    angle += 2.0 * PI;
  }

  return angle;
}


float wrapTo180(float angle) {
  while (angle > 180.0) {
    angle -= 360.0;
  }

  while (angle < -180.0) {
    angle += 360.0;
  }

  return angle;
}

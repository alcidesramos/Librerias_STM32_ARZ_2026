//funcion que escribe 8 BITS
void  I2C_Write(uint8_t value)
{
HAL_I2C_Master_Transmit(&hi2cs,SENSOR_ADDR ,&value, 1, 300);
}

//funcion que lee  8 BITS
uint8_t  I2C_Read()
{
uint8_t value;
HAL_I2C_Master_Receive(&hi2cs,SENSOR_ADDR ,&value, 1, 300);
return (value);
}

//ESCRIBE UN BYTE I2C en una posicion
void I2C_Writem(uint8_t dir,uint8_t val)
{
	HAL_I2C_Mem_Write(&hi2cs,SENSOR_ADDR, dir, I2C_MEMADD_SIZE_8BIT, &val, 1, 100);
}
